MVVM

Bind DocumentsSource, ToolsSource, ActiveContent, and SavedLayout, and the view model can drive the whole dock without touching the control. In ShellWindow the code-behind only calls InitializeComponent. ShellViewModel holds the documents, two tools, one toolbar, and the saved layout string, and never references DockManager. You don't need an MVVM framework: the sample uses a small RelayCommand, and the view models implement INotifyPropertyChanged.

<Window x:Class="SmartDock.Docs.Samples.ShellWindow"
        xmlns="http://schemas.microsoft.com/winfx/2006/xaml/presentation"
        xmlns:x="http://schemas.microsoft.com/winfx/2006/xaml"
        xmlns:dock="clr-namespace:SmartDock;assembly=SmartDock"
        xmlns:local="clr-namespace:SmartDock.Docs.Samples"
        Title="SmartDock"
        Height="640"
        Width="960">
  <Window.DataContext>
    <local:ShellViewModel />
  </Window.DataContext>
  <Window.Resources>
    <Style x:Key="DocumentItem" TargetType="dock:DockItem">
      <Setter Property="Title" Value="{Binding Title}" />
      <Setter Property="ContentId" Value="{Binding ContentId}" />
      <Setter Property="CanClose" Value="{Binding CanClose}" />
      <Setter Property="CanFloat" Value="{Binding CanFloat}" />
    </Style>
    <Style x:Key="ToolItem" TargetType="dock:DockItem">
      <Setter Property="Title" Value="{Binding Title}" />
      <Setter Property="ContentId" Value="{Binding ContentId}" />
      <Setter Property="CanClose" Value="{Binding CanClose}" />
      <Setter Property="CanFloat" Value="{Binding CanFloat}" />
      <Setter Property="CanAutoHide" Value="{Binding CanAutoHide}" />
      <Setter Property="HideOnClose" Value="{Binding HideOnClose}" />
    </Style>
    <Style x:Key="BarItem" TargetType="dock:DockToolBarItem">
      <Setter Property="Id" Value="{Binding Id}" />
      <Setter Property="Title" Value="{Binding Title}" />
      <Setter Property="Edge" Value="{Binding Edge}" />
    </Style>
    <DataTemplate x:Key="TextBody">
      <TextBox Text="{Binding Text, UpdateSourceTrigger=PropertyChanged}"
               AcceptsReturn="True"
               BorderThickness="0" />
    </DataTemplate>
    <DataTemplate x:Key="NoteBody">
      <TextBox Text="{Binding Text, UpdateSourceTrigger=PropertyChanged}"
               AcceptsReturn="True"
               BorderThickness="0"
               FontStyle="Italic" />
    </DataTemplate>
    <local:ShellTemplateSelector x:Key="Documents"
                                 Text="{StaticResource TextBody}"
                                 Note="{StaticResource NoteBody}" />
    <DataTemplate x:Key="ToolBody">
      <TextBlock Text="{Binding Text}" Margin="12" TextWrapping="Wrap" />
    </DataTemplate>
    <DataTemplate x:Key="ShellBar">
      <ToolBar>
        <Button Content="New" Command="{Binding NewDocument}" />
      </ToolBar>
    </DataTemplate>
  </Window.Resources>
  <DockPanel>
    <StackPanel DockPanel.Dock="Top" Orientation="Horizontal" Margin="4">
      <Button Content="Save layout"
              Margin="0,0,4,0"
              Command="{x:Static dock:DockCommands.SaveLayout}"
              CommandTarget="{Binding ElementName=Dock}" />
      <Button Content="Restore layout"
              Command="{x:Static dock:DockCommands.LoadLayout}"
              CommandTarget="{Binding ElementName=Dock}" />
    </StackPanel>
    <TextBlock DockPanel.Dock="Bottom" Margin="8,4" Text="{Binding ActiveTitle}" />
    <dock:DockManager x:Name="Dock"
                      DocumentItemStyle="{StaticResource DocumentItem}"
                      ToolItemStyle="{StaticResource ToolItem}"
                      DocumentTemplateSelector="{StaticResource Documents}"
                      ToolTemplate="{StaticResource ToolBody}"
                      ToolBarItemStyle="{StaticResource BarItem}"
                      ToolBarTemplate="{StaticResource ShellBar}"
                      DocumentsSource="{Binding Documents}"
                      ToolsSource="{Binding Tools}"
                      ToolBarsSource="{Binding ToolBars}"
                      ActiveContent="{Binding Active, Mode=TwoWay}"
                      SavedLayout="{Binding Layout}" />
  </DockPanel>
</Window>

Pane items

DocumentItemStyle and ToolItemStyle target DockItem, with the source item as the data context, so the setters bind straight to the view model:

using System.ComponentModel;
using System.Runtime.CompilerServices;

namespace SmartDock.Docs.Samples;

public abstract class DocumentViewModel : INotifyPropertyChanged
{
    private string _title;

    protected DocumentViewModel(string contentId, string title)
    {
        ContentId = contentId;
        _title = title;
    }

    public string ContentId { get; }

    public string Title
    {
        get => _title;
        set
        {
            _title = value;
            OnPropertyChanged();
        }
    }

    public bool CanClose { get; protected set; } = true;

    public bool CanFloat { get; protected set; } = true;

    public event PropertyChangedEventHandler? PropertyChanged;

    protected void OnPropertyChanged([CallerMemberName] string? name = null)
        => PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(name));
}

using System.ComponentModel;
using System.Runtime.CompilerServices;

namespace SmartDock.Docs.Samples;

public sealed class ToolViewModel : INotifyPropertyChanged
{
    private string _title;
    private string _text;

    public ToolViewModel(string contentId, string title, string text, bool canClose, bool canAutoHide, bool hideOnClose)
    {
        ContentId = contentId;
        _title = title;
        _text = text;
        CanClose = canClose;
        CanAutoHide = canAutoHide;
        HideOnClose = hideOnClose;
    }

    public string ContentId { get; }

    public string Title
    {
        get => _title;
        set
        {
            _title = value;
            OnPropertyChanged();
        }
    }

    public string Text
    {
        get => _text;
        set
        {
            _text = value;
            OnPropertyChanged();
        }
    }

    public bool CanClose { get; }

    public bool CanFloat => true;

    public bool CanAutoHide { get; }

    public bool HideOnClose { get; }

    public event PropertyChangedEventHandler? PropertyChanged;

    private void OnPropertyChanged([CallerMemberName] string? name = null)
        => PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(name));
}

DockItem has Title, ContentId, Icon, CanClose, CanFloat, CanAutoHide, and HideOnClose. A pane with an empty title shows its content id. An empty ContentId is replaced with item- followed by eight hex digits. A saved layout matches items by id, so these view models use fixed ids. Icon isn't saved in the layout file.

CanAutoHide defaults to true for tools whose style doesn't set it. This style binds it, so the view model decides. The close button appears when CanClose or HideOnClose is true. Solution Explorer sets CanClose to false and has no close button. Output sets HideOnClose, so closing it hides the pane and leaves the item in Tools.

When the user closes a pane, the manager removes the item from the source if the source is a writable IList, which ObservableCollection is. If the source isn't a list, the item stays, and the manager remembers that the user closed it, so resetting the same collection won't reopen it until the item is really removed.

For documents, DocumentTemplateSelector runs first. If it returns null, DocumentTemplate is used, and then the implicit DataTemplate for the item's type. ShellTemplateSelector returns the text template, the note template, or null:

using System.Windows;
using System.Windows.Controls;

namespace SmartDock.Docs.Samples;

public sealed class ShellTemplateSelector : DataTemplateSelector
{
    public DataTemplate? Text { get; set; }

    public DataTemplate? Note { get; set; }

    public override DataTemplate? SelectTemplate(object item, DependencyObject container)
    {
        if (item is TextDocumentViewModel)
            return Text;
        if (item is NoteDocumentViewModel)
            return Note;
        return null;
    }
}

Tools follow the same order with ToolTemplateSelector and ToolTemplate. This window sets ToolTemplate and no selector. Templates are ignored for items that are themselves a UIElement. Implicit templates in Window.Resources work too, since the manager looks them up again when it gets a parent and when it loads; Working in XAML uses that approach.

The manager reads the item style when an item is added and again whenever a bound property changes. Title raises PropertyChanged on the view model, so the tab text follows it.

Active pane

ActiveContent binds two-way by default. The manager writes it with SetCurrentValue, so your binding stays in place. Setting Active on the view model shows and activates that pane, even if it was hidden or auto-hidden. The value is null when the active pane didn't come from DocumentsSource or ToolsSource. ActiveTitle reads the same object, so the status line follows the selection.

If you set the active item before it's in a source, the manager keeps the value and activates the item once it appears.

Where new panes go

The sample never sets DockManager.Layout; the sources build the tree. The first pane of an empty manager becomes the root. A new document joins the main-tree document group that last held the active document. If there isn't one, it joins the first document group, and if there are no document groups, it goes to the right of the root at 0.75. A new tool joins the first tool group, or goes to the left of the root at 0.25. An item whose content id is already in the layout stays where the layout put it, whether that's a floating window, an auto-hide strip, or the hidden list.

NewDocument adds a TextDocumentViewModel with a new content id and sets Active, which activates the new pane:

private void AddDocument()
{
    _next++;
    var document = new TextDocumentViewModel("doc-" + _next, "Document " + _next, "");
    Documents.Add(document);
    Active = document;
}

Toolbars

ToolBarsSource builds one ToolBar per item, using ToolBarTemplateSelector, then ToolBarTemplate, then the implicit DataTemplate. The template's root must be a ToolBar; if it isn't, the manager ignores the template and creates an empty toolbar. The toolbar's data context is set to the item unless the toolbar already has one.

Placement comes from DockToolBarItem, not from the DockToolBar attached properties. ToolBarItemStyle targets DockToolBarItem and here binds Id, Title, and Edge. You can also set Band, BandIndex, Offset, Alignment, PaneId, IsVisible, ShowOverflow, and IsCustomizable. They're read when the toolbar is added and whenever a binding changes, but Edge and the band properties only place a toolbar that doesn't have a position yet. Toolbars explains each value.

using System.Windows.Input;

namespace SmartDock.Docs.Samples;

public sealed class ToolBarViewModel
{
    private readonly ShellViewModel _shell;

    public ToolBarViewModel(ShellViewModel shell) => _shell = shell;

    public string Id => "standard";

    public string Title => "Standard";

    public ToolBarEdge Edge => ToolBarEdge.Top;

    public ICommand NewDocument => _shell.NewDocument;
}

The New button binds to NewDocument on the toolbar view model, which forwards the shell's command, so it keeps working when the toolbar is torn off into its own window. ElementName bindings don't resolve inside the template.

Save and load

SavedLayout binds two-way by default, here to the view model's Layout string. The Save button runs DockCommands.SaveLayout with CommandTarget set to the manager, since the button is outside it; a button or menu item inside the manager doesn't need CommandTarget. The manager and each floating window handle the command. It writes the current layout to SavedLayout with SetCurrentValue, so the binding pushes the JSON into the view model's Layout property and the dock doesn't reload. If no layout has been assigned yet, the command does nothing, but by the time the window is shown the sources have already built one.

public string? Layout
{
    get => _layout;
    set
    {
        if (_layout == value)
            return;
        _layout = value;
        OnPropertyChanged();
    }
}

Any other change to SavedLayout loads the text exactly like LoadLayout. That includes a value set before the window is shown, so the first frame already uses it. Null and the empty string are ignored, but a string of spaces isn't empty. Text that LoadLayout can't read throws DockLayoutException and leaves the current tree alone. The SaveLayout method throws InvalidOperationException when no layout is assigned, which is why the command checks first.

The Restore button runs DockCommands.LoadLayout, with CommandTarget set to the manager. The command loads the current SavedLayout, so it puts the last save back even when that string has not changed. CanExecute is false when SavedLayout is null or empty, and the command does nothing in that case. Bad JSON throws DockLayoutException and leaves the tree as it is. A non-empty CommandParameter is loaded in place of SavedLayout, and that text is written back to SavedLayout without loading a second time.

The string is the same JSON described in Saving and loading layouts. It stores content ids, not view models. Loading replaces DockManager.Layout. A pane whose content id is still in Documents or Tools keeps its item, and a source item missing from the file opens at the default place described above. An id in the file that's no longer in a source and has no factory shows a "Missing content: id" placeholder. Because closing a document removes it from Documents, a layout saved before the close still names it.

RelayCommand is the entire command implementation, used by NewDocument.

using System.Windows.Input;

namespace SmartDock.Docs.Samples;

public sealed class RelayCommand : ICommand
{
    private readonly Action<object?> _execute;
    private readonly Predicate<object?>? _canExecute;

    public RelayCommand(Action<object?> execute, Predicate<object?>? canExecute = null)
    {
        _execute = execute;
        _canExecute = canExecute;
    }

    public bool CanExecute(object? parameter) => _canExecute?.Invoke(parameter) ?? true;

    public void Execute(object? parameter) => _execute(parameter);

    public event EventHandler? CanExecuteChanged
    {
        add => CommandManager.RequerySuggested += value;
        remove => CommandManager.RequerySuggested -= value;
    }
}
